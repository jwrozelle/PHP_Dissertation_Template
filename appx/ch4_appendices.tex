    \label{eq:pdf}
f_j(x_{0},y_{0}) = 
\begin{dcases}
\frac{1}{2\pi r_j\sqrt{x_{0}^{2}+y_{0}^{2}}}, & \text{if } 0 < \sqrt{x_{0}^2 + y_{0}^2} \leq r_j \\
